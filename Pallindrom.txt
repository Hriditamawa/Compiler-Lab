#include <stdio.h>

int main() {
char str[100];
int i, count = 0;

printf("Enter a string: ");
fgets(str, sizeof(str), stdin);

printf("\nPosition of spaces: ");

for (i = 0; str[i] != '\0'; i++) {
    if (str[i] == ' ') {
    printf("%d ", i);
    count++;
        }
    }

printf("\nNumber of spaces = %d\n", count);

for (i = 0; str[i] != '\0'; i++) {
    if (str[i] == ' ') {
    int j;

for (j = i; str[j] != '\0'; j++) {
    str[j] = str[j + 1];
            }
    i--;
    }
    }

    printf("String after removing spaces: %s", str);
    return 0;
}
